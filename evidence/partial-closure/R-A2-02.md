-- shop-trade 题目 01：仅创建 goods、order、payment 三张业务表。

-- 所有业务表使用 MySQL(InnoDB)：ENGINE=InnoDB、DEFAULT CHARSET=utf8mb4。
-- order 是 MySQL 保留字，必须使用反引号。

CREATE TABLE IF NOT EXISTS `goods` (
  `id` BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  `created_at` DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
  `updated_at` DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3) ON UPDATE CURRENT_TIMESTAMP(3),
  `deleted_at` DATETIME(3) NULL DEFAULT NULL,

  KEY `idx_goods_available_stock` (`available_stock`),

-- 并发风险点：下单与释放库存必须按 goods.id 主键命中聚簇索引并加 InnoDB 行锁；
-- 若更新条件未命中索引，InnoDB 可能扩大锁范围甚至升级为表锁，引发超卖风险。

CREATE TABLE IF NOT EXISTS `order` (

## 2. 缺口说明（本轮为何仍不能升级为 `pass`）

- 7 个演示账号来自 `server/api/v1/order.go` 中的 **controller 内存 map**（`demoAccounts`），不是数据库表中的记录；账号与角色**不落库**。
- `schema.sql` 中没有账号 / 用户表，本题也**不在授权补齐范围内**新增账号表（属题外数据表，需学生确认 C-02）。
- 本机无 MySQL 实例，即使落库也无从验证重启后仍可查。

## 3. 解除条件（可核查、可执行）

- 若题面/需求文档要求账号落库：经学生确认（C-02）后新增账号表与初始化数据，在真实 MySQL 上重启后复查 `/api/v1/me` 仍返回相同 role / user_id。
- 若题面不要求账号落库（当前口径）：由**验收 Agent 在独立会话**中确认「账号为演示级内存映射」符合题面「角色身份必须可信」的最小实现，本条按「不适用」结论归档，而非继续挂 `partial`。
