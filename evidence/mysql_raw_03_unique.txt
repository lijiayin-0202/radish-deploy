===== 阶段 4：唯一键冲突复现 =====
--- 4.1 写入订单基线数据 ---

--------------------------------------------------------------
$ mysql -uroot shop_trade -e "INSERT INTO `order` (order_no,buyer_id,goods_id,goods_no,goods_name,quantity,unit_price,total_amount,status,expires_at) VALUES ('UNIQ-PROBE-0001',1001,1,'GOODS-ACT-001','活动商品',1,10000,10000,'待支付',NOW(3)+INTERVAL 10 MINUTE) ON DUPLICATE KEY UPDATE order_no=VALUES(order_no);"
--------------------------------------------------------------
--- 4.2 重复插入相同 order_no：预期 uk_order_order_no 唯一约束报错 ---

--------------------------------------------------------------
$ mysql -uroot shop_trade -e "INSERT INTO `order` (order_no,buyer_id,goods_id,goods_no,goods_name,quantity,unit_price,total_amount,status,expires_at) VALUES ('UNIQ-PROBE-0001',1002,1,'GOODS-ACT-001','活动商品',1,10000,10000,'待支付',NOW(3)+INTERVAL 10 MINUTE);"
